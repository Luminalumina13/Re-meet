"use client";

import { useEffect, useState } from "react";
import { PREVIEW_PALETTES, type PreviewPalette } from "@/lib/dev/palettes";
import { cx } from "@/lib/format";

/**
 * Dev-only bar: repaints the whole app in a candidate palette (light and dark), so colours can be
 * judged on the real pages. It injects a stylesheet instead of inline styles, so the normal theme
 * toggle keeps working. Not shipped: layout renders it only when NODE_ENV is "development".
 */
const KEY = "remeet:preview-palette";
const STYLE_ID = "palette-preview-style";

function block(t: Record<string, string>, e: PreviewPalette["e"]): string {
  const v: Record<string, string> = {
    bg: t.bg, surface: t.surface, "surface-2": t.surface2, border: t.border, "border-strong": t.bs,
    text: t.text, "text-2": t.text2, "text-3": t.text3,
    accent: t.accent, "accent-hover": t.hover, "accent-strong": t.atext, "accent-text": t.atext, "accent-soft": t.soft, "accent-ink": t.ink,
    good: t.good, warn: t.warn, critical: t.crit,
    "series-1": t.s1, "series-2": t.s2, "series-3": t.s3, "series-4": t.s4, "series-5": t.s5, "series-6": t.s6,
    e1: e[0], e2: e[1], e3: e[2],
  };
  return Object.entries(v).map(([k, x]) => `--${k}:${x}`).join(";");
}

function apply(p: PreviewPalette | null) {
  document.getElementById(STYLE_ID)?.remove();
  if (!p) return;
  const el = document.createElement("style");
  el.id = STYLE_ID;
  el.textContent = `:root{${block(p.light, p.e)}}:root[data-theme="dark"]{${block(p.dark, p.e)}}`;
  document.head.appendChild(el);
}

export function PalettePreview() {
  const [active, setActive] = useState<string>("");
  const [open, setOpen] = useState(true);

  useEffect(() => {
    let saved = "";
    try {
      saved = localStorage.getItem(KEY) ?? "";
    } catch {
      /* storage unavailable */
    }
    const p = PREVIEW_PALETTES.find((x) => x.name === saved) ?? null;
    apply(p);
    // eslint-disable-next-line react-hooks/set-state-in-effect
    setActive(p?.name ?? "");
  }, []);

  const pick = (p: PreviewPalette | null) => {
    apply(p);
    setActive(p?.name ?? "");
    try {
      if (p) localStorage.setItem(KEY, p.name);
      else localStorage.removeItem(KEY);
    } catch {
      /* storage unavailable */
    }
  };

  if (!open) {
    return (
      <button type="button" onClick={() => setOpen(true)} className="fixed bottom-3 left-3 z-[100] rounded-full border border-border bg-surface px-3 py-1.5 text-xs font-semibold text-ink shadow-card">
        Палітри
      </button>
    );
  }
  const tab = (on: boolean) => cx("flex shrink-0 items-center gap-1.5 whitespace-nowrap rounded-lg px-2.5 py-1.5 text-xs font-semibold transition", on ? "bg-accent text-accent-ink" : "text-ink-2 hover:bg-surface-2");
  return (
    <div className="fixed inset-x-2 bottom-2 z-[100] mx-auto flex max-w-[98vw] items-center gap-1 rounded-2xl border border-border bg-surface/95 p-1.5 shadow-card backdrop-blur">
      <div className="flex min-w-0 flex-1 gap-1 overflow-x-auto [scrollbar-width:thin]">
        <button type="button" onClick={() => pick(null)} className={tab(active === "")}>
          Зараз
        </button>
        {PREVIEW_PALETTES.map((p) => (
          <button key={p.name} type="button" onClick={() => pick(p)} className={tab(active === p.name)} title={p.name}>
            <span className="inline-flex -space-x-1">
              {[p.light.accent, p.e[1], p.e[2]].map((c, i) => (
                <i key={i} className="h-3 w-3 rounded-full ring-1 ring-black/10" style={{ background: c }} />
              ))}
            </span>
            {p.name}
          </button>
        ))}
      </div>
      <button type="button" onClick={() => setOpen(false)} className="shrink-0 rounded-lg px-2 py-1.5 text-xs font-semibold text-ink-3 hover:bg-surface-2" aria-label="Сховати">
        ×
      </button>
    </div>
  );
}
